```typescript
// POST /api/daily-prices
export async function POST(request: NextRequest) {
  const { property_id, date, base_price } = await request.json()
  
  const { data } = await supabase
    .from('daily_prices')
    .upsert({ property_id, date, base_price })
    .select()
    .single()
  
  return json({ success: true, data })
}

// PUT /api/daily-prices/bulk
export async function PUT(request: NextRequest) {
  const { property_id, dates, base_price } = await request.json()
  
  const rows = dates.map((date: string) => ({
    property_id,
    date,
    base_price
  }))
  
  await supabase
    .from('daily_prices')
    .upsert(rows)
  
  return json({ success: true, updated: rows.length })
}

// DELETE /api/daily-prices
export async function DELETE(request: NextRequest) {
  const { property_id, date } = await request.json()
  
  await supabase
    .from('daily_prices')
    .delete()
    .eq('property_id', property_id)
    .eq('date', date)
  
  return json({ success: true })
}
```

### Component

```tsx
// src/components/DailyPricesManager.tsx

export function DailyPricesManager({ propertyId, month }: Props) {
  const [prices, setPrices] = useState<Map<string, number>>(new Map())
  const [loading, setLoading] = useState(false)
  const [undoStack, setUndoStack] = useState<any[]>([])

  const handlePriceChange = async (date: string, price: number) => {
    setUndoStack([...undoStack, { date, price: prices.get(date) }])
    setPrices(new Map(prices).set(date, price))
    
    const response = await fetch('/api/daily-prices', {
      method: 'POST',
      body: JSON.stringify({ property_id: propertyId, date, base_price: price })
    })
    
    if (!response.ok) {
      alert('Erro ao atualizar preço')
    }
  }

  const handleBulkFill = async (price: number) => {
    if (!window.confirm(`Preencher todos os dias com €${price}?`)) return
    
    const dates = eachDayOfInterval({
      start: startOfMonth(new Date()),
      end: endOfMonth(new Date())
    }).map(d => format(d, 'yyyy-MM-dd'))

    setLoading(true)
    const response = await fetch('/api/daily-prices/bulk', {
      method: 'PUT',
      body: JSON.stringify({
        property_id: propertyId,
        dates,
        base_price: price
      })
    })
    setLoading(false)
    
    if (response.ok) {
      alert(`Atualizado ${dates.length} dias`)
    }
  }

  const handleUndo = () => {
    if (undoStack.length === 0) return
    
    const last = undoStack[undoStack.length - 1]
    setUndoStack(undoStack.slice(0, -1))
    handlePriceChange(last.date, last.price)
  }

  return (
    <div className="space-y-4 p-4">
      <h3 className="font-bold">Preços por Dia</h3>
      
      <div className="flex gap-2">
        <input
          type="number"
          placeholder="Preço"
          className="px-2 py-1 border rounded"
          id="bulk-price"
        />
        <button
          onClick={() => handleBulkFill(parseFloat((document.getElementById('bulk-price') as HTMLInputElement).value))}
          disabled={loading}
          className="px-3 py-1 bg-blue-600 text-white rounded disabled:opacity-50"
        >
          Preencher Tudo
        </button>
        <button
          onClick={handleUndo}
          disabled={undoStack.length === 0}
          className="px-3 py-1 border rounded disabled:opacity-50"
        >
          Undo
        </button>
      </div>
      
      <div className="grid grid-cols-7 gap-2">
        {/* Render 28-31 dias do mês */}
        {Array.from({ length: 31 }).map((_, i) => {
          const date = format(addDays(startOfMonth(new Date()), i), 'yyyy-MM-dd')
          return (
            <input
              key={date}
              type="number"
              value={prices.get(date) || ''}
              onChange={(e) => handlePriceChange(date, parseFloat(e.target.value))}
              placeholder={format(parseISO(date), 'd')}
              className="p-2 border rounded text-xs text-center"
            />
          )
        })}
      </div>
    </div>
  )
}
```
