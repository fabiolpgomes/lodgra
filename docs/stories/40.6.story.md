## Implementation

### Calendar Cell Update

```tsx
// src/components/calendar/CalendarCell.tsx

interface CalendarCellProps {
  date: Date
  reservation?: Reservation
  dailyPrice?: number
  currency: string
}

export function CalendarCell({
  date,
  reservation,
  dailyPrice,
  currency
}: CalendarCellProps) {
  const formatPrice = (price: number) => {
    switch (currency) {
      case 'EUR':
        return `€${price.toFixed(0)}`
      case 'BRL':
        return `R$${price.toFixed(0)}`
      case 'USD':
        return `$${price.toFixed(0)}`
      default:
        return `${price.toFixed(0)}`
    }
  }

  return (
    <div className="p-2 border min-h-[80px]">
      <div className="text-sm font-semibold">{format(date, 'd')}</div>
      
      {reservation && (
        <div className="mt-1 bg-blue-100 p-1 rounded text-xs">
          <div className="font-medium">{reservation.guest_name}</div>
          <div className="text-gray-600">{reservation.nights}n</div>
        </div>
      )}
      
      {dailyPrice && !reservation && (
        <div className="mt-2 text-xs text-gray-600 font-medium hover:text-gray-900">
          {formatPrice(dailyPrice)}
        </div>
      )}
    </div>
  )
}
```
