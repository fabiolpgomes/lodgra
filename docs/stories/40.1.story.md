#### POST /api/admin/review/[id]/decision — Submeter Decisão

**Request:**
```json
{
  "token": "xxx",
  "decision": "APPROVED" | "PARTIAL" | "DENIED",
  "refund_percentage": 100 | 50 | 0,
  "notes": "Confirmado problema de limpeza"
}
```

**Response (200):**
```json
{
  "success": true,
  "reservation_id": "uuid",
  "status": "REFUNDED" | "DENIED",
  "refund_amount": 450.00,
  "refund_id": "re_xxx"
}
```

**Backend Logic:**
```typescript
export async function POST(request: NextRequest) {
  const { reservation_id, token, decision, refund_percentage, notes } = await request.json()
  
  // 1. Validar token
  const reservation = await db.reservations.findUnique({ where: { id: reservation_id } })
  if (reservation.review_token !== token || reservation.review_token_expires_at < new Date()) {
    return error('Token inválido ou expirado', 401)
  }
  
  // 2. Validar status
  if (reservation.status !== 'PENDING_MANUAL_REVIEW') {
    return error('Reserva não está em manual review', 400)
  }
  
  // 3. Calcular refund
  const refund_amount = (reservation.total_amount * refund_percentage) / 100
  
  // 4. Se APPROVED/PARTIAL, criar refund no Stripe
  if (['APPROVED', 'PARTIAL'].includes(decision)) {
    const refund = await stripePT.refunds.create({
      charge: reservation.stripe_charge_id,
      amount: Math.round(refund_amount * 100),
      reason: 'serious_issue_resolved',
    })
    
    await db.refunds.create({
      reservation_id,
      stripe_refund_id: refund.id,
      amount: refund_amount,
      refund_percentage,
      reason: 'serious_issue',
      policy_applied: { manual_review: true },
    })
  }
  
  // 5. Atualizar reservation
  await db.reservations.update({
    where: { id: reservation_id },
    data: {
      status: decision === 'APPROVED' || decision === 'PARTIAL' ? 'REFUNDED' : 'DENIED',
      refund_amount: ['APPROVED', 'PARTIAL'].includes(decision) ? refund_amount : 0,
      manual_review_notes: notes,
      reviewed_by: adminUserId,
      reviewed_at: new Date(),
    }
  })
  
  // 6. Enviar emails
  await sendEmail(reservation.guest_email, {
    subject: 'Seu caso foi revisado',
    html: decisionEmailTemplate({
      decision,
      refund_amount,
      notes
    })
  })
  
  await sendEmail('ahspropriedades@gmail.com', {
    subject: 'Caso resolvido',
    html: confirmationEmailTemplate({ reservation_id, decision })
  })
  
  return { success: true, status: decision === 'APPROVED' || decision === 'PARTIAL' ? 'REFUNDED' : 'DENIED' }
}
```

```tsx
// /app/admin/review/[id]/page.tsx
export default async function ReviewPage({ params, searchParams }: Props) {
  const { token } = searchParams
  
  const response = await fetch(
    `/api/admin/review/${params.id}?token=${token}`,
    { headers: { 'Cookie': cookies().toString() } }
  )
  
  if (!response.ok) {
    return <ErrorState message="Link inválido ou expirado" />
  }
  
  const data = await response.json()
  
  return (
    <div className="min-h-screen bg-gray-50 p-4">
      <div className="max-w-md mx-auto bg-white rounded-lg shadow-sm p-6">
        <h1 className="text-2xl font-bold mb-2">Revisar Caso</h1>
        <p className="text-gray-600 mb-6">Expira em {data.expires_in_hours}h</p>
        
        <div className="space-y-4 mb-6">
          <div className="bg-gray-50 p-4 rounded">
            <p className="text-sm text-gray-600">Hóspede</p>
            <p className="font-semibold">{data.guest_name}</p>
          </div>
          <div className="bg-gray-50 p-4 rounded">
            <p className="text-sm text-gray-600">Propriedade</p>
            <p className="font-semibold">{data.property_name}</p>
          </div>
          <div className="bg-yellow-50 p-4 rounded border border-yellow-200">
            <p className="text-sm text-gray-600">Problema Reportado</p>
            <p className="font-semibold">{data.description}</p>
          </div>
        </div>

          token={token}

// ReviewDecisionForm component
function ReviewDecisionForm({ reservationId, token, totalAmount }) {
  const [decision, setDecision] = useState<'APPROVED' | 'PARTIAL' | 'DENIED'>('PARTIAL')
  const [notes, setNotes] = useState('')
  const [loading, setLoading] = useState(false)
  
  const refundPercentage = {
    'APPROVED': 100,
    'PARTIAL': 50,
    'DENIED': 0
  }[decision]
  
  const refundAmount = (totalAmount * refundPercentage) / 100
  
  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    setLoading(true)
    
    const response = await fetch(`/api/admin/review/${reservationId}/decision`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        token,
        decision,
        refund_percentage: refundPercentage,
        notes
      })
    })
    
    if (response.ok) {
      alert('Decisão registada com sucesso!')
    } else {
      alert('Erro ao processar decisão')
    }
    setLoading(false)
  }
  
  return (
    <form onSubmit={handleSubmit} className="space-y-4">
      <div>
        <label className="block text-sm font-medium mb-2">Decisão</label>
        <div className="space-y-2">
          {['APPROVED', 'PARTIAL', 'DENIED'].map(d => (
            <label key={d} className="flex items-center">
              <input
                type="radio"
                name="decision"
                value={d}
                checked={decision === d}
                onChange={(e) => setDecision(e.target.value as any)}
                className="mr-2"
              />
              <span className="text-sm">
                {d === 'APPROVED' && '✅ Reembolso Completo (100%)'}
                {d === 'PARTIAL' && '⚠️ Reembolso Parcial (50%)'}
                {d === 'DENIED' && '❌ Sem Reembolso'}
              </span>
            </label>
          ))}
        </div>
      </div>
      
      <div className="bg-blue-50 p-3 rounded text-sm">
        Reembolso: <strong>€{refundAmount.toFixed(2)}</strong>
      </div>
      
      <div>
        <label className="block text-sm font-medium mb-2">Notas (Opcional)</label>
        <textarea
          value={notes}
          onChange={(e) => setNotes(e.target.value)}
          placeholder="Motivo da decisão..."
          className="w-full p-2 border rounded text-sm"
          rows={3}
        />
      </div>
      
      <button
        type="submit"
        disabled={loading}
        className="w-full bg-blue-600 text-white py-2 rounded font-medium disabled:opacity-50"
      >
        {loading ? 'Processando...' : 'Registar Decisão'}
      </button>
    </form>
  )
}
```
