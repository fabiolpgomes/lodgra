## Technical Spec

### POST /api/reservations/create

```typescript
// src/app/api/reservations/create/route.ts

export async function POST(request: NextRequest) {
  try {
    const {
      property_id,
      check_in,
      check_out,
      guest_name,
      guest_email,
      guest_phone,
      number_of_guests,
      notes,
      payment_method = 'card'
    } = await request.json()

    // 1. Validações básicas
    const errors: string[] = []
    if (!property_id) errors.push('property_id requerido')
    if (!check_in || !check_out) errors.push('Datas requeridas')
    if (!guest_email) errors.push('Email requerido')
    
    if (errors.length > 0) {
      return json({ success: false, errors }, { status: 400 })
    }

    // 2. Fetch property + policy
    const supabase = createAdminClient()
    const { data: property } = await supabase
      .from('properties')
      .select('id, currency, min_nights, slug')
      .eq('id', property_id)
      .single()

    if (!property) {
      return json({ error: 'Propriedade não encontrada' }, { status: 404 })
    }

    const { data: policy } = await supabase
      .from('property_cancellation_policies')
      .select('*')
      .eq('property_id', property_id)
      .eq('is_active', true)
      .single()

    // 3. Validar datas
    const checkInDate = parseISO(check_in)
    const checkOutDate = parseISO(check_out)
    const nights = differenceInDays(checkOutDate, checkInDate)

    if (nights < (property.min_nights || 1)) {
      return json({
        success: false,
        error: `Mínimo de ${property.min_nights || 1} noites requeridas`
      }, { status: 400 })
    }

    // 4. Validar disponibilidade (calendar_blocks)
    const { data: blockedDates } = await supabase
      .from('calendar_blocks')
      .select('id')
      .eq('property_id', property_id)
      .gte('date', check_in)
      .lt('date', check_out)
      .limit(1)

    if (blockedDates && blockedDates.length > 0) {
      return json({
        success: false,
        error: 'Datas não disponíveis'
      }, { status: 409 })
    }

    // 5. Calcular preço
    const priceResponse = await fetch(
      `${process.env.NEXT_PUBLIC_APP_URL}/api/reservations/calculate-price`,
      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ property_id, check_in, check_out })
      }
    )
    const { total_amount } = await priceResponse.json()

    if (!total_amount || total_amount <= 0) {
      return json({ error: 'Erro ao calcular preço' }, { status: 400 })
    }

    // 6. Criar Stripe Payment Intent
    const stripeKey = getStripeKey(property.currency)
    const stripe = new Stripe(stripeKey, { apiVersion: '2026-02-25.clover' })

    const paymentIntent = await stripe.paymentIntents.create({
      amount: Math.round(total_amount * 100), // cents
      currency: property.currency.toLowerCase(),
      payment_method_types: ['card'],
      metadata: {
        property_id,
        reservation_purpose: 'guest_booking'
      }
    })

    // 7. Criar reservation em DB
    const { data: reservation } = await supabase
      .from('reservations')
      .insert({
        property_id,
        guest_name,
        guest_email,
        guest_phone,
        check_in,
        check_out,
        number_of_guests,
        notes,
        status: 'PENDING',
        payment_status: 'unpaid',
        total_amount,
        currency: property.currency,
        stripe_payment_intent_id: paymentIntent.id,
        cancellation_policy_id: policy?.id,
        cancellation_policy_snapshot: policy ? {
          policy_type: policy.policy_type,
          full_refund_days: policy.full_refund_days,
          partial_refund_days: policy.partial_refund_days,
          partial_refund_percent: policy.partial_refund_percent,
          captured_at: new Date().toISOString()
        } : null,
        source: 'direct_booking'
      })
      .select('id')
      .single()

    return json({
      success: true,
      reservation_id: reservation.id,
      client_secret: paymentIntent.client_secret,
      total_amount,
      currency: property.currency
    })
  } catch (error) {
    console.error('[reservations/create] Error:', error)
    return json({
      error: 'Erro ao criar reserva'
    }, { status: 500 })
  }
}
```

### POST /api/stripe/payment-intent/confirm

```typescript
// src/app/api/stripe/payment-intent/confirm/route.ts

export async function POST(request: NextRequest) {
  try {
    const { reservation_id, payment_method_id } = await request.json()

    // 1. Fetch reservation
    const supabase = createAdminClient()
    const { data: reservation } = await supabase
      .from('reservations')
      .select('*')
      .eq('id', reservation_id)
      .single()

    if (!reservation) {
      return json({ error: 'Reservação não encontrada' }, { status: 404 })
    }

    // 2. Confirmar payment intent no Stripe
    const stripeKey = getStripeKey(reservation.currency)
    const stripe = new Stripe(stripeKey, { apiVersion: '2026-02-25.clover' })

    const confirmed = await stripe.paymentIntents.confirm(
      reservation.stripe_payment_intent_id,
      { payment_method: payment_method_id }
    )

    if (confirmed.status !== 'succeeded') {
      return json({
        success: false,
        error: 'Pagamento falhou',
        status: confirmed.status
      }, { status: 402 })
    }

    // 3. Bloquear datas no calendário (imediato)
    const dates = eachDayOfInterval({
      start: parseISO(reservation.check_in),
      end: addDays(parseISO(reservation.check_out), -1)
    })

    await supabase
      .from('calendar_blocks')
      .insert(
        dates.map(date => ({
          property_id: reservation.property_id,
          date: format(date, 'yyyy-MM-dd'),
          reservation_id: reservation.id,
          status: 'blocked'
        }))
      )

    // Nota: Status atualiza para CONFIRMED via webhook (charge.succeeded)
    // Mas podemos atualizar aqui também para UX instantânea
    await supabase
      .from('reservations')
      .update({
        status: 'CONFIRMED',
        payment_status: 'paid',
        stripe_charge_id: confirmed.charges.data[0]?.id
      })
      .eq('id', reservation_id)

    // 4. Enviar emails
    await sendEmail(reservation.guest_email, {
      subject: 'Reserva Confirmada',
      template: 'reservation-confirmation',
      data: { reservation }
    })

    await sendEmail(process.env.MANAGER_EMAIL, {
      subject: `Nova Reserva: ${reservation.guest_name}`,
      template: 'manager-new-reservation',
      data: { reservation }
    })

    return json({
      success: true,
      reservation_id,
      status: 'CONFIRMED'
    })
  } catch (error) {
    console.error('[stripe/confirm] Error:', error)
    return json({ error: 'Erro ao confirmar pagamento' }, { status: 500 })
  }
}
```

### Webhook: charge.succeeded

```typescript
// src/app/api/webhooks/stripe/route.ts (extend)

export async function POST(request: NextRequest) {
  const signature = request.headers.get('stripe-signature')
  const body = await request.text()

  try {
    const event = stripe.webhooks.constructEvent(
      body,
      signature,
      process.env.STRIPE_WEBHOOK_SECRET
    )

    if (event.type === 'charge.succeeded') {
      const charge = event.data.object
      const metadata = charge.metadata

      if (metadata.reservation_purpose === 'guest_booking') {
        const supabase = createAdminClient()
        
        // Atualizar reservation
        await supabase
          .from('reservations')
          .update({
            status: 'CONFIRMED',
            payment_status: 'paid',
            stripe_charge_id: charge.id
          })
          .eq('stripe_payment_intent_id', charge.payment_intent)
      }
    }

    return json({ success: true })
  } catch (error) {
    console.error('[webhooks/stripe] Error:', error)
    return json({ error: 'Webhook error' }, { status: 400 })
  }
}
```
