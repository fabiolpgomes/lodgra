// e2e/booking-complete-flow.spec.ts

test('Complete booking flow EUR → confirmation', async ({ page }) => {
  await page.goto('/property/test-property/calendar')
  
  // Select dates
  await page.click('[data-test-id="day-15"]')
  await page.click('[data-test-id="day-20"]')
  
  // Check price display
  await expect(page.locator('[data-test-id="total-price"]')).toContainText('€')
  
  // Fill form
  await page.fill('[name="guest_name"]', 'Test Guest')
  await page.fill('[name="guest_email"]', 'test@example.com')
  
  // Payment
  await page.click('[data-test-id="pay-button"]')
  await page.waitForURL('**/payment')
  
  // Fill card (Stripe test)
  const frame = page.frameLocator('iframe[name="__stripe"]')
  await frame.locator('[placeholder="Card number"]').fill('4242 4242 4242 4242')
  await frame.locator('[placeholder="MM"]').fill('12')
  await frame.locator('[placeholder="YY"]').fill('25')
  await frame.locator('[placeholder="CVC"]').fill('123')
  
  // Confirm
  await page.click('[data-test-id="confirm-payment"]')
  
  // Success
  await expect(page.locator('[data-test-id="success-message"]')).toBeVisible()
  await expect(page.locator('[data-test-id="reservation-id"]')).toContainText(/^[a-f0-9-]+$/)
})

test('Voluntary cancellation with refund', async ({ page }) => {
  // Create reservation first
  const reservationId = await createTestReservation()
  
  // Navigate to calendar
  await page.goto(`/property/test-property/calendar?res=${reservationId}`)
  
  // Open cancellation modal
  await page.click('[data-test-id="cancel-button"]')
  
  // Select "sou cedo"
  await page.click('[data-test-id="cancel-voluntary"]')
  
  // Fill description
  await page.fill('[name="description"]', 'Change of plans due to illness')
  
  // Check refund estimate
  const refund = await page.locator('[data-test-id="refund-amount"]').textContent()

  await page.click('[data-test-id="confirm-cancel"]')

  await expect(page.locator('[data-test-id="cancel-success"]')).toBeVisible()
})

// e2e/performance.spec.ts

test('Booking checkout < 2s response time', async ({ page }) => {
  const startTime = performance.now()
  
  await page.goto('/property/test-property/calendar')
  await page.click('[data-test-id="day-15"]')
  await page.click('[data-test-id="day-20"]')
  await page.waitForTimeout(100)
  
  const endTime = performance.now()
  expect(endTime - startTime).toBeLessThan(2000)
})

test('Calendar render < 1s', async ({ page }) => {

  await page.goto('/property/test-property/calendar')
  await page.waitForLoadState('networkidle')
  
  const endTime = performance.now()
  expect(endTime - startTime).toBeLessThan(1000)
})
