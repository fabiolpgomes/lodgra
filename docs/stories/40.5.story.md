### Stripe Config

```typescript
// src/lib/stripe/utils.ts

export function getStripeKey(currency: string): string {
  const c = currency.toUpperCase()
  
  switch (c) {
    case 'BRL':
      return (process.env.STRIPE_BR_SECRET_KEY || '').trim()
    case 'EUR':
      return (process.env.STRIPE_PT_SECRET_KEY || '').trim()
    case 'USD':
      return (process.env.STRIPE_SECRET_KEY || '').trim()
    default:
      throw new Error(`Unsupported currency: ${c}`)
  }
}

export function isValidCurrency(currency: string): boolean {
  return ['EUR', 'BRL', 'USD'].includes(currency.toUpperCase())
}
```

### API Updates

```typescript
// src/app/api/reservations/create/route.ts

const VALID_CURRENCIES = ['EUR', 'BRL', 'USD']

if (!VALID_CURRENCIES.includes(property.currency)) {
  return json({
    success: false,
    error: `Invalid currency: ${property.currency}`
  }, { status: 400 })
}

const stripeKey = getStripeKey(property.currency)
if (!stripeKey) {
  return json({
    success: false,
    error: `Stripe not configured for ${property.currency}`
  }, { status: 500 })
}
```
